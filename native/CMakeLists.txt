cmake_minimum_required(VERSION 3.20)
project(LifeEngineNative VERSION 1.0.0 LANGUAGES CXX RC)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(life_engine_native_core WIN32 src/main.cpp)
add_executable(life_engine WIN32 src/skia_preview.cpp src/simulation.cpp src/atlas.cpp src/app_support.cpp src/skia_runtime.rc)
add_custom_target(life_engine_skia_preview DEPENDS life_engine)
add_executable(life_engine_native_benchmark EXCLUDE_FROM_ALL benchmarks/simulation_benchmark.cpp src/simulation.cpp src/atlas.cpp)
target_include_directories(life_engine_native_benchmark PRIVATE src)
target_compile_options(life_engine_native_benchmark PRIVATE -O3 -DNDEBUG -Wall -Wextra)
add_executable(life_engine_ecology_benchmark EXCLUDE_FROM_ALL benchmarks/ecology_benchmark.cpp src/simulation.cpp src/atlas.cpp)
target_include_directories(life_engine_ecology_benchmark PRIVATE src)
target_compile_options(life_engine_ecology_benchmark PRIVATE -O3 -DNDEBUG -Wall -Wextra)
configure_file(vendor/skia/x64/libSkiaSharp.dll ${CMAKE_CURRENT_BINARY_DIR}/libSkiaSharp.dll COPYONLY)
target_compile_options(life_engine_native_core PRIVATE -O3 -Wall -Wextra)
target_link_options(life_engine_native_core PRIVATE -municode -static -static-libgcc -static-libstdc++)
target_link_libraries(life_engine_native_core PRIVATE user32 gdi32)
target_compile_options(life_engine PRIVATE -O3 -Wall -Wextra)
target_link_options(life_engine PRIVATE -municode -static -static-libgcc -static-libstdc++)
target_link_libraries(life_engine PRIVATE user32 gdi32 winmm comdlg32)
add_custom_command(TARGET life_engine POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    ${CMAKE_CURRENT_SOURCE_DIR}/vendor/skia/x64/libSkiaSharp.dll
    $<TARGET_FILE_DIR:life_engine>/libSkiaSharp.dll
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:life_engine>
    $<TARGET_FILE_DIR:life_engine>/life_engine_skia_preview.exe)
if(BUILD_TESTING)
  add_executable(life_engine_ui_tests tests/ui_commands_tests.cpp)
  target_include_directories(life_engine_ui_tests PRIVATE src)
  add_test(NAME native_ui_commands COMMAND life_engine_ui_tests)
  add_test(NAME native_ui_headless COMMAND life_engine "--headless-script=${CMAKE_CURRENT_SOURCE_DIR}/tests/ui_smoke.commands")
  set_tests_properties(native_ui_headless PROPERTIES TIMEOUT 45)
  add_executable(life_engine_optimization_tests tests/optimization_tests.cpp src/simulation.cpp src/atlas.cpp)
  target_include_directories(life_engine_optimization_tests PRIVATE src)
  target_compile_options(life_engine_optimization_tests PRIVATE -O2)
  add_test(NAME native_optimization_contracts COMMAND life_engine_optimization_tests)
  add_executable(life_engine_frame_tests tests/frame_exchange_tests.cpp src/simulation.cpp src/atlas.cpp)
  target_include_directories(life_engine_frame_tests PRIVATE src)
  target_compile_options(life_engine_frame_tests PRIVATE -O2)
  add_test(NAME native_frame_exchange COMMAND life_engine_frame_tests)
  add_test(NAME native_follow_headless COMMAND life_engine "--headless-script=${CMAKE_CURRENT_SOURCE_DIR}/tests/follow_smoke.commands")
  set_tests_properties(native_follow_headless PROPERTIES TIMEOUT 90)
  add_executable(life_engine_watch_tests tests/watch_tests.cpp src/simulation.cpp src/atlas.cpp)
  target_include_directories(life_engine_watch_tests PRIVATE src)
  target_compile_options(life_engine_watch_tests PRIVATE -O2)
  add_test(NAME native_watch_replay COMMAND life_engine_watch_tests)
  add_executable(life_engine_depth_tests tests/depth_tests.cpp src/simulation.cpp src/atlas.cpp)
  target_include_directories(life_engine_depth_tests PRIVATE src)
  target_compile_definitions(life_engine_depth_tests PRIVATE LEGACY_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_compile_options(life_engine_depth_tests PRIVATE -O2)
  add_test(NAME native_depth_save_replay COMMAND life_engine_depth_tests)
  add_executable(life_engine_native_tests tests/reference_tests.cpp src/simulation.cpp src/atlas.cpp src/app_support.cpp)
  target_include_directories(life_engine_native_tests PRIVATE src)
  target_compile_options(life_engine_native_tests PRIVATE -O2 -Wall -Wextra)
  target_link_libraries(life_engine_native_tests PRIVATE bcrypt)
  add_test(NAME native_reference_parity COMMAND life_engine_native_tests)
endif()
